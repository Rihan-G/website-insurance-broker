      CREATE TABLE IF NOT EXISTS rate_sheets (
        id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
        insurer TEXT NOT NULL,
        product TEXT NOT NULL,
        version TEXT NOT NULL,
        file_path TEXT NOT NULL,          -- path in Supabase Storage
        file_name TEXT NOT NULL,
        file_size TEXT,
        uploaded_by UUID REFERENCES profiles(id),
        created_at TIMESTAMPTZ DEFAULT now()
      );

      ALTER TABLE rate_sheets ENABLE ROW LEVEL SECURITY;
      -- Only admin/broker can read or write
      CREATE POLICY "staff read rate_sheets" ON rate_sheets
        FOR SELECT USING (
          EXISTS (SELECT 1 FROM profiles WHERE id = auth.uid() AND role IN ('admin','broker'))
        );
      CREATE POLICY "staff insert rate_sheets" ON rate_sheets
        FOR INSERT WITH CHECK (
          EXISTS (SELECT 1 FROM profiles WHERE id = auth.uid() AND role IN ('admin','broker'))
        );
      CREATE POLICY "staff delete rate_sheets" ON rate_sheets
        FOR DELETE USING (
          EXISTS (SELECT 1 FROM profiles WHERE id = auth.uid() AND role IN ('admin','broker'))
        );

    - New storage bucket: rate_sheets (private, staff only)
      Add to Supabase Dashboard > Storage > New bucket > name: rate-sheets, private: true
      Or add to the migration above:
        INSERT INTO storage.buckets (id, name, public) VALUES ('rate-sheets', 'rate-sheets', false);

    - Update RateSheetsPage.tsx:
        - Upload: use supabase.storage.from('rate-sheets').upload(path, file)
        - List: SELECT * FROM rate_sheets ORDER BY created_at DESC
        - Download: use supabase.storage.from('rate-sheets').createSignedUrl(file_path, 3600)
        - Delete: DELETE FROM rate_sheets WHERE id = ? + storage.remove([file_path])
    - Replace: DEMO_SHEETS array with live Supabase query
